LocalScript

A basic example Luau weather script.

📅 09/05/2026 
local Lighting = game:GetService("Lighting")

local function Day()
    Lighting.ClockTime = 14
    Lighting.Brightness = 2
end

local function Night()
    Lighting.ClockTime = 0
    Lighting.Brightness = 0.5
end

while true do
    Day()
    task.wait(30)

    Night()
    task.wait(30)
end
